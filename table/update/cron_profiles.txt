-- เช็คและเพิ่มคอลัมน์ที่ขาด
-- sql

DELIMITER $$

CREATE PROCEDURE update_cron_profiles_safe()
BEGIN

    -- id
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='cron_profiles' AND COLUMN_NAME='id'
    ) THEN
        ALTER TABLE cron_profiles 
            ADD COLUMN id INT(11) NOT NULL AUTO_INCREMENT PRIMARY KEY FIRST;
    END IF;

    -- label
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='cron_profiles' AND COLUMN_NAME='label'
    ) THEN
        ALTER TABLE cron_profiles 
            ADD COLUMN label VARCHAR(100) NULL AFTER id;
    END IF;

    -- cron_expr
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='cron_profiles' AND COLUMN_NAME='cron_expr'
    ) THEN
        ALTER TABLE cron_profiles 
            ADD COLUMN cron_expr VARCHAR(30) NOT NULL AFTER label;
    END IF;

    -- description
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='cron_profiles' AND COLUMN_NAME='description'
    ) THEN
        ALTER TABLE cron_profiles 
            ADD COLUMN description MEDIUMTEXT NULL AFTER cron_expr;
    END IF;

    -- notify_mode
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='cron_profiles' AND COLUMN_NAME='notify_mode'
    ) THEN
        ALTER TABLE cron_profiles 
            ADD COLUMN notify_mode ENUM('FAIL_ONLY','ALL') NOT NULL DEFAULT 'ALL' AFTER description;
    END IF;

END$$

DELIMITER ;

-- รัน procedure
CALL update_cron_profiles_safe();
