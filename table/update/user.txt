DELIMITER $$

CREATE PROCEDURE update_user_safe()
BEGIN

    -- id
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='user' AND COLUMN_NAME='id'
    ) THEN
        ALTER TABLE user 
            ADD COLUMN id INT(11) NOT NULL AUTO_INCREMENT PRIMARY KEY FIRST;
    END IF;

    -- username
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='user' AND COLUMN_NAME='username'
    ) THEN
        ALTER TABLE user 
            ADD COLUMN username VARCHAR(50) NOT NULL AFTER id;
    END IF;

    -- password
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='user' AND COLUMN_NAME='password'
    ) THEN
        ALTER TABLE user 
            ADD COLUMN password VARCHAR(255) NOT NULL AFTER username;
    END IF;

    -- role
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='user' AND COLUMN_NAME='role'
    ) THEN
        ALTER TABLE user 
            ADD COLUMN role ENUM('user','admin') NULL DEFAULT 'user' AFTER password;
    END IF;

    -- active
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='user' AND COLUMN_NAME='active'
    ) THEN
        ALTER TABLE user 
            ADD COLUMN active TINYINT(1) NULL DEFAULT 1 AFTER role;
    END IF;

    -- twofa_secret
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='user' AND COLUMN_NAME='twofa_secret'
    ) THEN
        ALTER TABLE user 
            ADD COLUMN twofa_secret VARCHAR(64) NULL AFTER active;
    END IF;

    -- twofa_enabled
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_NAME='user' AND COLUMN_NAME='twofa_enabled'
    ) THEN
        ALTER TABLE user 
            ADD COLUMN twofa_enabled TINYINT(1) NULL DEFAULT 0 AFTER twofa_secret;
    END IF;

    -- Unique Index: username
    IF NOT EXISTS (
        SELECT 1 FROM INFORMATION_SCHEMA.STATISTICS
        WHERE TABLE_NAME='user' AND INDEX_NAME='username'
    ) THEN
        ALTER TABLE user 
            ADD UNIQUE INDEX username (username);
    END IF;

END$$

DELIMITER ;

-- รัน procedure
CALL update_user_safe();
